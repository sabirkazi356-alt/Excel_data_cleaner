function applyCleaningPipeline() { if (!rawFileRows || rawFileRows.length === 0) return; const detectServiceType = document.getElementById('optDetectServiceType').checked; const extractDocStatus = document.getElementById('optExtractDocStatus').checked; const explodeMultiline = document.getElementById('optExplodeMultiline').checked; const cleanNoise = document.getElementById('optCleanNoise').checked; const fixDates = document.getElementById('optFixDates').checked; const dedupe = document.getElementById('optDedupe').checked; // Behtar header row detection jo aap ki file ke mutabiq row dhoond le gi let headerRowIndex = rawFileRows.findIndex(row => row.some(cell => { let val = String(cell).toLowerCase().trim(); return val === 'id' || val === 'name' || val === 'designation' || val === 'document'; })); if (headerRowIndex === -1) headerRowIndex = 0; let rawHeaders = rawFileRows[headerRowIndex].map(h => String(h).trim()); let dataRows = rawFileRows.slice(headerRowIndex + 1); let parsedRecords = []; let origCount = dataRows.length; dataRows.forEach(row => { let record = {}; rawHeaders.forEach((h, idx) => { if (h !== "" && h !== "undefined") { let val = row[idx] !== undefined ? row[idx] : ""; if (typeof val === 'string' && cleanNoise && val.startsWith("#######")) val = ""; record[h] = val; } }); if (Object.keys(record).length === 0) return; // Document column ko dynamically dhoondna chahe naam kuch bhi ho let rawDocText = ""; Object.keys(record).forEach(k => { if (/document|doc/i.test(k)) { if (record[k]) rawDocText = record[k]; } }); let docEntries = parseDocumentEntries(rawDocText); if (explodeMultiline && docEntries.length > 1) { docEntries.forEach(entry => { let newRec = { ...record }; if (extractDocStatus) { newRec['Document'] = entry.document; newRec['Document Status'] = entry.status; } if (detectServiceType) { newRec['Service'] = detectServiceCategory(entry.document); } parsedRecords.push(newRec); }); } else { if (extractDocStatus) { record['Document'] = docEntries[0].document; record['Document Status'] = docEntries[0].status; } if (detectServiceType) { record['Service'] = detectServiceCategory(docEntries[0].document); } parsedRecords.push(record); } });
